% Presenter notes are embedded with \presenternotes below.
% Select exactly one mode. Overrides may follow \input{presenter-notes.tex}.
%   normal deck:             \PresenterNormaltrue
%   presenter notes only:    \PresenterNotesOnlytrue
%   slides with notes right: \PresenterNotesRighttrue
\newif\ifPresenterNormal
\newif\ifPresenterNotesOnly
\newif\ifPresenterNotesRight
\PresenterNormalfalse
\PresenterNotesOnlyfalse
\PresenterNotesRighttrue

% Delay the original option setup so per-output overrides can follow \input.
% pgfpages is loaded in the preamble; it is used only by notes-right mode.
\usepackage{pgfpages}
% A compact heading leaves room for the full script; the default thumbnail
% consumes a quarter of a widescreen notes page.
\setbeamerfont{note page}{size=\fontsize{9}{10}\selectfont}
\setbeamertemplate{note page}{%
  \begingroup
  \vspace*{6pt}
  \textbf{Slide \insertframenumber: \insertshortframetitle}\par\smallskip
  \insertnote
  \endgroup
}
\newcount\PresenterModeCount
\AtBeginDocument{%
  \PresenterModeCount=0\relax
  \ifPresenterNormal \advance\PresenterModeCount by 1\relax\fi
  \ifPresenterNotesOnly \advance\PresenterModeCount by 1\relax\fi
  \ifPresenterNotesRight \advance\PresenterModeCount by 1\relax\fi
  \ifnum\PresenterModeCount=1\relax\else
    \PackageError{presenter-notes}{Select exactly one presenter mode}%
      {Set one Presenter switch true and the other two false.}%
  \fi
  \ifPresenterNormal
    \setbeameroption{hide notes}
  \fi
  \ifPresenterNotesOnly
    \setbeameroption{show only notes}
  \fi
  \ifPresenterNotesRight
    \setbeameroption{show notes on second screen=right}
  \fi
}

\newcommand{\presenternotes}[2]{%
\note{%
  \csname textbf\endcsname{Point plan}
\begin{itemize}
#1
\end{itemize}
\medskip
  \csname textbf\endcsname{Readable version}
\par #2
}%
}
